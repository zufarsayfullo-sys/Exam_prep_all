function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); var timestamp = new Date(); var name = data.name || "Anonymous"; var percentage = data.percentage !== undefined ? data.percentage + "%" : "N/A"; var mistakes = data.mistakes || "None"; // Header format: Timestamp | Student Name | Percentage | Mistakes sheet.appendRow([timestamp, name, percentage, mistakes]); return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT); } catch (err) { return ContentService.createTextOutput("Error: " + err.toString()).setMimeType(ContentService.MimeType.TEXT); } }
